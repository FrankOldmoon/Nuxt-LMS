Answer <- Radius ^ 2

// --- 6. 数组（一维 / 二维声明与访问）---

// --- 7. 输入与输出（INPUT / OUTPUT 多值逗号分隔）---

// --- 8. 选择：IF / ELSE / ENDIF（THEN 单独一行）---
IF ChallengerScore > ChampionScore
  THEN
    OUTPUT ChallengerName, " is the new champion"
  ELSE
    OUTPUT ChampionName, " is still the champion"

// --- 9. 选择：CASE OF / OTHERWISE / ENDCASE ---
INPUT Move
CASE OF Move
  'W' : Position <- Position - 10
  'E' : Position <- Position + 10
  'A' : Position <- Position - 1
  'D' : Position <- Position + 1
  OTHERWISE OUTPUT "Beep"
ENDCASE

// --- 10. 迭代：FOR ... TO ... [STEP] ... NEXT ---
FOR Row <- 1 TO 10
    OUTPUT Row
NEXT Row
FOR Index <- 10 TO 1 STEP -1
    OUTPUT Index
NEXT Index

// --- 11. 迭代：REPEAT ... UNTIL（后测）---
REPEAT
    OUTPUT "Please enter the password"
    INPUT Password
UNTIL Password = "Secret"

// --- 12. 迭代：WHILE ... DO ... ENDWHILE（前测）---
WHILE Number > 9 DO
    Number <- Number - 9
ENDWHILE

// --- 13. 过程：定义与调用（PROCEDURE / CALL，无参可无括号）---
PROCEDURE Line(Size : INTEGER)
    DECLARE Length : INTEGER
    FOR Length <- 1 TO Size
        OUTPUT '-'
    NEXT Length
ENDPROCEDURE

PROCEDURE DefaultLine
    CALL Line(60)
ENDPROCEDURE

CALL DefaultLine
CALL Line(MySize)

// --- 14. 函数：定义与调用（FUNCTION ... RETURNS / RETURN）---
FUNCTION SumSquare(Number1 : INTEGER, Number2 : INTEGER) RETURNS INTEGER
    RETURN Number1 * Number1 + Number2 * Number2
ENDFUNCTION
OUTPUT "Sum of squares = ", SumSquare(10, 20)
